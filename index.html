noBtn.addEventListener("click", function () { // Show the next cute message const message = messages[noClickCount % messages.length]; showMessage(message); // Change button text noBtn.textContent = buttonTexts[Math.min(noClickCount, buttonTexts.length - 1)]; // Move the button somewhere else moveNoButton(); // Make YES button slightly bigger const currentScale = 1 + Math.min(noClickCount * 0.04, 0.25); yesBtn.style.transform = `scale(${currentScale})`; noClickCount++; }); function moveNoButton() { // Make the button independent of the normal layout noBtn.style.position = "fixed"; // Button dimensions const buttonWidth = noBtn.offsetWidth; const buttonHeight = noBtn.offsetHeight; // Safe area around the screen edges const padding = 25; // Available screen space const maxX = window.innerWidth - buttonWidth - padding; const maxY = window.innerHeight - buttonHeight - padding; // Generate completely random coordinates const randomX = padding + Math.random() * (maxX - padding); const randomY = padding + Math.random() * (maxY - padding); // Smooth movement noBtn.style.transition = "left 0.4s ease, top 0.4s ease, transform 0.2s ease"; noBtn.style.left = randomX + "px"; noBtn.style.top = randomY + "px"; // Little bounce noBtn.style.transform = "scale(1.15)"; setTimeout(() => { noBtn.style.transform = "scale(1)"; }, 200); }
